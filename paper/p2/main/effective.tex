\section{Computability and the finite form}\label{sec:eff}

\begingroup\sloppy

This section proves Theorem~\ref{thm:main}(iii). The output, a rational polynomial times a fixed smooth spatial cutoff, is found by searching finite certificates. A candidate is accepted when Euler trajectories from a grid covering the sources stay where the cutoff equals one and end inside the target, with an error bound valid on $[0,1]$. Accepted candidates are correct (\Sref{Theorem}{thm:certificate-sound}). Termination uses the Bernstein compilation of $H[n,k,C,c,s]$ (\S\ref{sec:eff-finite}); any packing Hamiltonian with a positive margin would serve, and for the Epoch family the witness is Corollary~\ref{thm:main-epoch} at density $1-\eps/2$. A name for a real $x$ is a rational stream $q(p)$ with $|q(p)-x|\le2^{-p}$; rationals are named by constant streams. Names of $r_i=\sqrt{c_i/\pi}$ and $R=\sqrt{C/\pi}$ are computed in \Ssec{sec:efd}. Here $|\cdot|$ is the Euclidean norm and $S$ is the flat step of \Sref{Definition}{I:def:pre-flat}.

\subsection{The finite form}\label{sec:eff-finite}

\begin{lemma}[Gr\"onwall]\label{lem:eff-gronwall}
\uses{S-lem:cpl-gronwall-abstract}
Let $G,H\colon\R\times\R^{2n}\to\R$ be $C^\infty$, let $\eta>0$ and $L_X>0$, and suppose that
\begin{enumerate}[label=(\roman*)]
\item $|X_G(t,z)-X_H(t,z)|\le\eta$ for all $t\in[0,1]$ and all $z\in\R^{2n}$;
\item $X_H(t,\cdot)$ is $L_X$-Lipschitz on all of $\R^{2n}$ for every $t\in[0,1]$.
\end{enumerate}
If $\gamma_G,\gamma_H\colon[0,1]\to\R^{2n}$ are integral curves of $X_G$ and $X_H$ with
$\gamma_G(0)=\gamma_H(0)$, then $|\gamma_G(1)-\gamma_H(1)|\le\eta(e^{L_X}-1)/L_X$. If moreover
$\eta=b_Re^{-L_X}/2$ with $b_R>0$, then $\eta(e^{L_X}-1)/L_X<b_R$.
\end{lemma}

\begin{proof}
$u=\gamma_G-\gamma_H$ satisfies $u(0)=0$ and $|\dot u|\le\eta+L_X|u|$, so Gr\"onwall gives the bound. With $\eta=b_Re^{-L_X}/2$ it equals $\tfrac{b_R}2(1-e^{-L_X})/L_X<b_R/2$ (cf.\ \Sref{Lemma}{lem:cpl-gronwall}).
\end{proof}

For smooth compactly supported $H$, (ii) holds for every $L_X>0$ with $L_X\ge\max_{[0,1]\times\R^{2n}}\|D^2_zH\|$, as $X_H=J_0\nabla_zH$ with $J_0$ orthogonal.

Let $n\ge2$, $k\ge1$, $C>0$, $c\in(0,\infty)^k$, $s\in(\R^{2n})^k$, and let $H$ satisfy $\Pack(n,k,C,c,s)$ (Definition~\ref{def:intro-spec}): $H$ is $C^\infty$, vanishes outside $\R\times K_H$ with $K_H$ compact, and $\phi^1_H(B(s_i,c_i))\subset\Bo(C)$. Write $\zeta_1,\dots,\zeta_{2n}$ for the coordinates of $z$. A \emph{compilation datum} for $H$ consists of:

\begin{enumerate}[label=(C\arabic*)]
\item $C_0\in(0,C)$ with $\phi^1_H(B(s_i,c_i))\subset B(C_0)$ for all $i$; put $R_*=\sqrt{C_0/\pi}<R_C=\sqrt{C/\pi}$.
\item A rational $a_{\rm box}>0$ with $H(t,z)=0$ whenever $\max_l|\zeta_l|\ge a_{\rm box}$, and an integer $B>\max(2n,2a_{\rm box})$; the box cutoff is
 \[
 \beta_B(z)=\prod_{l=1}^{2n}\beta_{B,1}(\zeta_l),\qquad
 \beta_{B,1}(x)=1-S\Bigl(\frac{x^2-(B/2)^2}{(3B/4)^2-(B/2)^2}\Bigr),
 \]
 so $0\le\beta_B\le1$, $\beta_B=1$ on $[-B/2,B/2]^{2n}$, $\beta_B=0$ outside $(-3B/4,3B/4)^{2n}$, and $\beta_BH=H$.
\item Reals $L_X\ge1+\sup_{[0,1]\times\R^{2n}}\|D^2_zH\|$ and $C_\beta\ge1+\sup_{\R^{2n}}|\nabla\beta_B|$.
\item A rational $b_R$ with $0<b_R<(R_C-R_*)/4$; put $\eta=b_Re^{-L_X}/2$.
\item A real $M_0\ge1$ bounding all first and second partials on $[0,1]^{2n+1}$ of $\tilde H(\xi)=H(\xi_0,2B\xi_1-B,\dots,2B\xi_{2n}-B)$.
\item A rational $\delta$ with $0<\delta<\min\bigl(\tfrac12,\eta/(16(1+C_\beta))\bigr)$ and an integer $N_{\rm B}>\max\bigl(1,((2n+1)M_0/\delta)^2,4M_0/\delta\bigr)$.
\item For every $m\in\{0,\dots,N_{\rm B}\}^{2n+1}$ a rational $\tilde H_m$ with $|\tilde H_m-\tilde H(m/N_{\rm B})|<\delta/(2N_{\rm B}+1)$.
\end{enumerate}
With $b_{N,j}(x)=\binom Njx^j(1-x)^{N-j}$ put
\[
 P(t,z)=\sum_{m\in\{0,\dots,N_{\rm B}\}^{2n+1}}\tilde H_m\,b_{N_{\rm B},m_0}(t)\prod_{l=1}^{2n}b_{N_{\rm B},m_l}\Bigl(\frac{\zeta_l+B}{2B}\Bigr),
 \qquad \Hfin=\beta_B\cdot P,
\]
so $P\in\Q[t,z]$ has degree at most $N_{\rm B}$ in each variable. Compilation data exist: $\bigcup_i\phi^1_H(B(s_i,c_i))$ is a compact subset of $\Bo(C)$, $H$ has compact support, (C3), (C5) are maxima of continuous functions on compacta, and the other items are strict bounds on single numbers.

\begin{theorem}[Finite output]\label{thm:eff-finite}
\uses{lem:eff-gronwall, def:intro-spec, S-thm:cpl-final, S-prop:cpl-bernstein, S-lem:cpl-cutoff}
Let $H$ satisfy $\Pack(n,k,C,c,s)$. For every compilation datum (C1)--(C7) for $H$, the function
$\Hfin=\beta_B\cdot P$ satisfies $\Pack(n,k,C,c,s)$. More precisely: $\Hfin$ is $C^\infty$ and $\Hfin(t,z)=0$ for
all $t$ whenever $z\notin(-3B/4,3B/4)^{2n}$; $|X_{\Hfin}-X_H|<\eta$ on $[0,1]\times\R^{2n}$; and for every
$z\in\bigcup_iB(s_i,c_i)$,
\[
 |\phi^1_{\Hfin}(z)-\phi^1_H(z)|<b_R,\qquad |\phi^1_{\Hfin}(z)|<R_*+b_R<R_C,
\]
so that $\phi^1_{\Hfin}$ maps every $B(s_i,c_i)$ into $\Bo(C)$. If the source balls are pairwise disjoint,
their images are pairwise disjoint.
\end{theorem}

\begin{proof}[Proof (complete version: \Sref{Theorem}{thm:cpl-final})]
The tensor Bernstein estimates (\Sref{Proposition}{prop:cpl-bernstein}) and the rounding (C7), harmless since Bernstein weights are nonnegative with sum $1$, give $|P-H|<\delta$ and $|\nabla_zP-\nabla_zH|<\delta$ on $[0,1]\times[-B,B]^{2n}$ (\Sref{Lemma}{lem:cpl-poly}). As $\nabla_z\Hfin-\nabla_zH=\beta_B\nabla_z(P-H)+(P-H)\nabla\beta_B$ there, its norm is $<\delta(1+C_\beta)<\eta/16$ by (C6); outside $(-B,B)^{2n}$ both vanish nearby. Hence $|X_{\Hfin}-X_H|<\eta$, $X_H$ is $L_X$-Lipschitz by (C3), the flows are global, and Lemma~\ref{lem:eff-gronwall} with (C1) gives the estimates. Support holds as $\beta_B$ is independent of $t$; disjointness follows from injectivity of $\phi^1_{\Hfin}$.
\end{proof}

\subsection{The algorithm}\label{sec:eff-algorithm}


Put $d=2n$. A candidate consists of integers $B,Q,N\ge1$, precisions $p,m\ge0$, and a rational arithmetic expression $e$ for a polynomial $P_e(t,z)$. The integer $\Theta=h_B(e)$ (\Sref{Definition}{def:certificate-program}) bounds, in maximum norm, the speed and the temporal and spatial Lipschitz constants of
\[
 F_e(t,z)=J_0\nabla_zP_e(\operatorname{clip}_{[0,1]}t,
                  \operatorname{clip}_{[-B/2,B/2]^d}z),
\]
$\operatorname{clip}_X$ being nearest-point projection to the box $X$. Read the input names to error $2^{-p}$; form rational outer balls $B(s_i^p,r^+_{i,p})$ containing the closed sources and a rational $R^-_p\le\sqrt{C/\pi}$ (\Sref{Equation}{eq:certificate-enclosures}); test $|s^p_{i,l}|+r^+_{i,p}<Q$ for all $i,l$. On the grid of mesh $\rho=2Q2^{-m}$ in $[-Q,Q]^d$, discard (node $a$, source $i$) if $|a-s_i^p|^2>(r^+_{i,p}+d\rho)^2$; otherwise compute the exact Euler table
\[
 x_0=a,\quad x_{j+1}=x_j+N^{-1}F_e(j/N,x_j),\quad
 E=\left(\rho+\frac{\Theta+\Theta^2}{N}\right)4^\Theta+\frac{\Theta}{N},
\]
and require $|x_{j,l}|+E<B/2$ for all $j\le N$ and $l$, $R^-_p-dE>0$ and $|x_N|^2<(R^-_p-dE)^2$. These are finite strict rational tests; the algorithm returns $B$ and $P_e\in\Q[t,z]$ for the first candidate passing all of them. The real inputs are oracles and $n,k$ ordinary input (\Ssec{sec:alg}). The returned polynomial need not be the Bernstein approximation of $H[n,k,C,c,s]$: Bernstein compilation only shows that some candidate is accepted, and every accepted candidate is correct.

\begin{theorem}[Termination and correctness]\label{thm:eff-algorithm}
There is one oracle machine with the following property. Let $n\ge2$, $k\ge1$, let $(n,k,C,c)$ be admissible and let
the closed balls $B(s_i,c_i)\subset\R^{2n}$ be pairwise disjoint (these are the hypotheses of
Theorem~\ref{thm:main}). Supply as oracles names of $C,c_1,\dots,c_k$ and of the $2nk$ coordinates
of $s_1,\dots,s_k$. On input $(n,k)$ the machine halts and outputs an integer $B$ and a polynomial
$P\in\Q[t,z_1,\ldots,z_{2n}]$ such that $\beta_BP$ satisfies $\Pack(n,k,C,c,s)$. The output may depend on the names,
not only on the numbers they name. If the names are computable, for instance the constant names of rational
inputs, replacing the oracles by programs for them gives an ordinary algorithm.

There is also one oracle machine which, given a name of $\eps\in(0,1)$ as oracle and an integer $k\ge9$ as input,
computes names of the source radius $1$ and of the target radius $R=(k/(1-\eps/2))^{1/4}$ for the data
\[
 n=2,\quad C=\pi R^2,\quad c_i=\pi,\quad s_i=(3i-3,0,0,0),
\]
runs the same certificate search on them, and halts with $(B,P)$ such that $\beta_BP$ satisfies $\Pack(2,k,C,c,s)$;
here $k/R^4=1-\eps/2>1-\eps$.
\leanmark{original\_name\_algorithm, epoch\_name\_algorithm, epoch\_name\_algorithm\_packing}
\end{theorem}

\begin{proof}
\emph{Correctness.} Each source point is within $\rho$ (maximum norm) of a retained node and the Euler error is at most $E$, so the plateau tests keep the $F_e$-trajectory where $F_e=X_{\beta_BP_e}$ and the terminal test ends it in the open target (\Sref{Theorem}{thm:certificate-sound}).

\emph{Termination.} $H[n,k,C,c,s]$ maps the sources into $\Bo(C_0)$, $C_0<C$. Theorem~\ref{thm:eff-finite} with target $(C_0+C)/2$ and $B$ large gives a rational polynomial whose trajectories from a neighbourhood of the sources stay a positive distance inside $\{\beta_B=1\}$ and end a positive distance inside $\Bo(C)$; as retained nodes lie within $(2d+3)2^{-p}+d\rho$ of a source, all tests pass for $p,m,N$ large (\Sref{Theorem}{thm:certificate-exists}). Each candidate uses finitely many queries and operations, and $n,k$ are inputs of one machine (\Ssec{sec:alg}).

\emph{Epoch case.} The input is admissible with disjoint sources (Corollary~\ref{thm:main-epoch}); radius names come from $\eps$ by total programs: the reciprocal of $1-\eps/2>\frac12$ and two square roots (\Sref{Lemma}{lem:efd-roots-direct}).
\end{proof}

\begin{remark}[Uniformity and size]\label{rem:eff-branches}\label{rem:eff-size}
The exact construction distinguishes the four cases of Construction~\ref{cons:ham-branch}, but the finite verifier
has the same form in every case: it tests rational trajectories of the proposed output, and needs no equality test
between named capacities. The factor $4^\Theta$ in the Euler bound can require very large grids; the theorem concerns
computability, not running time.
\end{remark}

\begin{remark}[The Epoch submission format]\label{rem:eff-epoch-interface}
Epoch \cite{Epoch} asks for a Python function returning a PyTorch implementation of the Hamiltonian, which a numerical
verifier integrates from sampled boundary points of the source balls within fixed time limits. We read its variables as
$(Q,P)=(x,y)$, with $X_H=(\partial_PH,-\partial_QH)$, and its target as the time-one map. The construction and the algorithm are designed for proof and computability, not for integration within the
verifier's limits of ten minutes per solution and ten seconds per evaluation. The repository includes two
accompanying programs in \texttt{program/}: one for the Epoch Hamiltonian interface, based on the elementary
constructions of the third paper of the release~\cite{SongP3} over a covered range of $(k,\eps)$, and one evaluating
closed-form $C^4$ packing embeddings and exact-real inverse certificates directly for all $k\ge10$ and $\eps\in(0,1)$,
proved in the fourth paper of the release~\cite{SongP4}.
\end{remark}
